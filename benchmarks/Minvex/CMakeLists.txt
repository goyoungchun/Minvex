cmake_minimum_required(VERSION 3.24)
if(NOT TARGET minvex::viewer)
    project(MinvexBenchmark LANGUAGES CXX)
    find_package(Minvex 1.0 CONFIG REQUIRED COMPONENTS viewer graphics)
endif()
add_executable(minvex_benchmark Main.cpp)
target_link_libraries(minvex_benchmark PRIVATE minvex::viewer)
add_executable(minvex_ui_frame_probe UiFrameProbe.cpp)
target_link_libraries(minvex_ui_frame_probe PRIVATE minvex::viewer)
if(MSVC)
    target_compile_options(minvex_ui_frame_probe PRIVATE /utf-8)
endif()
