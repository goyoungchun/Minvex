cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0141)
    cmake_policy(SET CMP0141 NEW)
    # Keep compiler probes independent of the MSVC compiler PDB service.
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
endif()
project(MinvexQuickStart LANGUAGES CXX)

option(MINVEX_QUICKSTART_BUILD_EXTRAS "Build the additional SDK examples" ON)
set(_minvex_components viewer graphics)
if(MINVEX_QUICKSTART_BUILD_EXTRAS)
    list(APPEND _minvex_components embedded)
    set(_minvex_optional OPTIONAL_COMPONENTS physics)
endif()
find_package(Minvex 1.1.0 CONFIG REQUIRED COMPONENTS ${_minvex_components} ${_minvex_optional})
include("${Minvex_DIR}/MinvexMSVCDependencies.cmake" OPTIONAL)
add_executable(minvex_quickstart Main.cpp)
target_link_libraries(minvex_quickstart PRIVATE minvex::viewer)
if(NOT MINVEX_QUICKSTART_BUILD_EXTRAS)
    return()
endif()
add_executable(minvex_quickstart_procedural Procedural.cpp)
target_link_libraries(minvex_quickstart_procedural PRIVATE minvex::viewer)
add_executable(minvex_quickstart_skinning Skinning.cpp)
target_link_libraries(minvex_quickstart_skinning PRIVATE minvex::viewer)
add_executable(minvex_fitting Fitting.cpp)
target_link_libraries(minvex_fitting PRIVATE minvex::viewer)
add_executable(minvex_maintenance Maintenance.cpp)
target_link_libraries(minvex_maintenance PRIVATE minvex::viewer)
add_executable(minvex_survey Survey.cpp)
target_link_libraries(minvex_survey PRIVATE minvex::viewer)
if(Minvex_physics_FOUND)
    add_executable(minvex_load_planning LoadPlanning.cpp)
    target_link_libraries(minvex_load_planning PRIVATE minvex::viewer minvex::physics)
endif()
if(MSVC)
    target_compile_options(minvex_maintenance PRIVATE /utf-8)
endif()

if(APPLE)
    enable_language(OBJCXX)
    set_source_files_properties(FrameCapture.cpp PROPERTIES LANGUAGE OBJCXX)
endif()
add_executable(minvex_quickstart_capture FrameCapture.cpp)
target_link_libraries(minvex_quickstart_capture PRIVATE minvex::embedded)
if(WIN32)
    target_compile_definitions(minvex_quickstart_capture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(minvex_quickstart_capture PRIVATE d3d11)
elseif(APPLE)
    set_target_properties(minvex_quickstart_capture PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
    target_compile_options(minvex_quickstart_capture PRIVATE -fobjc-arc)
    target_link_libraries(minvex_quickstart_capture PRIVATE "-framework Cocoa" "-framework Metal")
endif()
