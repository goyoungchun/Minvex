cmake_minimum_required(VERSION 3.24)
project(MinvexInstalledGraphicsChecks LANGUAGES CXX)
find_package(Minvex 1.0 CONFIG REQUIRED COMPONENTS embedded graphics)
include("${Minvex_DIR}/MinvexMSVCDependencies.cmake" OPTIONAL)
get_filename_component(_repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(APPLE)
    enable_language(OBJCXX)
endif()
foreach(_case "lighting|Lighting/LightingPixels.cpp" "optimization|Optimization/GpuEfficiencyTests.cpp" "ui|Ui/UiPixels.cpp" "lifecycle|Rendering/NativeLifecycleTests.cpp")
    string(REPLACE "|" ";" _pair "${_case}")
    list(GET _pair 0 _name)
    list(GET _pair 1 _source)
    set(_source "${_repo}/tests/${_source}")
    set(_target "minvex_installed_${_name}")
    add_executable(${_target} "${_source}")
    target_link_libraries(${_target} PRIVATE minvex::embedded)
    target_compile_features(${_target} PRIVATE cxx_std_20)
    target_include_directories(${_target} PRIVATE "${_repo}/tests")
    target_compile_definitions(${_target} PRIVATE MINVEX_FIXTURE_DIR="${_repo}/tests/Fixtures" MINVEX_TEST_OUTPUT_DIR="${CMAKE_CURRENT_BINARY_DIR}/output")
    if(APPLE)
        set_source_files_properties("${_source}" PROPERTIES LANGUAGE OBJCXX)
        set_target_properties(${_target} PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
        target_compile_options(${_target} PRIVATE -fobjc-arc)
        target_link_libraries(${_target} PRIVATE "-framework Cocoa" "-framework Metal")
    elseif(WIN32)
        target_compile_options(${_target} PRIVATE /utf-8)
        target_compile_definitions(${_target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(${_target} PRIVATE d3d11)
    endif()
endforeach()
